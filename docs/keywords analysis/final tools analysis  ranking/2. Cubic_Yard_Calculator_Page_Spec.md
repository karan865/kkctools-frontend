## B4. Reference calculation engine (tested)

Tested with Node 22 — all 36 checks in B5 pass.

```js
// volume-engine.js — shared "dimensions -> volume" core (no DOM, no deps).
// Supports: rectangle, circle, triangle (prism), area x depth, and known volume.
// Cubic Yard page uses this directly. Asphalt / Stone / Sand / Fill Dirt can reuse it later.

const FT_PER_M = 3.280839895013123;
const KG_PER_LB = 0.45359237;
const LB_PER_SHORT_TON = 2000;
const EPS = 1e-9;
const ceilSafe = (x) => Math.ceil(x - EPS);          // avoids 45.0000000001 -> 46

const LENGTH_TO_FT = { ft: 1, in: 1 / 12, yd: 3, m: FT_PER_M, cm: FT_PER_M / 100, mm: FT_PER_M / 1000 };
const AREA_TO_SQFT = { sqft: 1, sqyd: 9, sqm: FT_PER_M ** 2 };
const VOLUME_TO_FT3 = { ft3: 1, yd3: 27, m3: FT_PER_M ** 3, in3: 1 / 1728, l: (FT_PER_M ** 3) / 1000 };
const DENSITY_TO_LB_FT3 = { 'lb/ft3': 1, 'kg/m3': 1 / 16.01846337396, 'tons/yd3': LB_PER_SHORT_TON / 27, 'lb/yd3': 1 / 27 };

// Typical bagged concrete mix yields (ft3 per bag). Always tell users to check the bag label.
const BAG_YIELD_FT3 = { 40: 0.30, 50: 0.375, 60: 0.45, 80: 0.60 };
const CONCRETE_DENSITY_LB_FT3 = 150;                    // normal-weight concrete

const toFt = (v, u) => v * LENGTH_TO_FT[u];

function sectionVolumeFt3(s) {
  const qty = s.qty == null ? 1 : s.qty;
  let v;
  switch (s.shape) {
    case 'rectangle': v = toFt(s.length, s.lengthUnit) * toFt(s.width, s.widthUnit) * toFt(s.depth, s.depthUnit); break;
    case 'circle': { const r = toFt(s.diameter, s.diameterUnit) / 2; v = Math.PI * r * r * toFt(s.depth, s.depthUnit); break; }
    case 'triangle': v = 0.5 * toFt(s.base, s.baseUnit) * toFt(s.height, s.heightUnit) * toFt(s.depth, s.depthUnit); break;
    case 'area': v = s.area * AREA_TO_SQFT[s.areaUnit] * toFt(s.depth, s.depthUnit); break;
    case 'volume': v = s.volume * VOLUME_TO_FT3[s.volumeUnit]; break;
    default: throw new Error('Unknown shape: ' + s.shape);
  }
  return v * qty;
}

function validate(input) {
  const errors = [];
  const pos = (n) => typeof n === 'number' && isFinite(n) && n > 0;
  const secs = input.sections || [];
  if (!secs.length) errors.push('Add at least one section.');
  if (secs.length > 10) errors.push('Maximum 10 sections.');
  secs.forEach((s, i) => {
    const q = s.qty == null ? 1 : s.qty;
    if (!(Number.isInteger(q) && q >= 1 && q <= 1000)) errors.push(`Section ${i + 1}: quantity must be a whole number from 1 to 1,000.`);
    let v = NaN; try { v = sectionVolumeFt3(s); } catch (e) {}
    if (!pos(v)) errors.push(`Section ${i + 1}: enter dimensions greater than 0.`);
  });
  if (!(input.wastePct >= 0 && input.wastePct <= 50)) errors.push('Waste must be between 0% and 50%.');
  if (input.density != null && input.density !== '') {
    const d = input.density * DENSITY_TO_LB_FT3[input.densityUnit || 'lb/ft3'];
    if (!pos(d)) errors.push('Enter a density greater than 0.');
  }
  return errors;
}

function calcVolume(input) {
  const errors = validate(input);
  if (errors.length) return { ok: false, errors };

  const material = input.material || 'concrete';
  const breakdown = input.sections.map((s, i) => {
    const ft3 = sectionVolumeFt3(s);
    return { label: s.label || `Section ${i + 1}`, ft3, yd3: ft3 / 27 };
  });
  const exactFt3 = breakdown.reduce((a, b) => a + b.ft3, 0);
  const factor = 1 + input.wastePct / 100;
  const vols = (f) => ({ ft3: exactFt3 * f, yd3: (exactFt3 * f) / 27, m3: (exactFt3 * f) / (FT_PER_M ** 3) });
  const exact = vols(1), order = vols(factor);

  const out = {
    ok: true, material, breakdown, exact, order,
    suggestedOrderYd3: ceilSafe(order.yd3 * 4) / 4,          // round UP to nearest 1/4 yd3 (ready-mix style)
  };

  // Weight: concrete defaults to 150 lb/ft3; other materials only if a density is supplied
  let dens = null;
  if (input.density != null && input.density !== '') dens = input.density * DENSITY_TO_LB_FT3[input.densityUnit || 'lb/ft3'];
  else if (material === 'concrete') dens = CONCRETE_DENSITY_LB_FT3;
  if (dens) {
    const w = (ft3) => { const lb = ft3 * dens; return { pounds: lb, tons: lb / LB_PER_SHORT_TON, tonnes: lb * KG_PER_LB / 1000 }; };
    out.densityLbFt3 = dens;
    out.weightExact = w(exactFt3);
    out.weightOrder = w(exactFt3 * factor);
  }

  if (material === 'concrete') {
    out.bags = Object.keys(BAG_YIELD_FT3).map((k) => ({
      sizeLb: Number(k), yieldFt3: BAG_YIELD_FT3[k],
      bags: ceilSafe(order.ft3 / BAG_YIELD_FT3[k]),
    }));
    out.guidance = order.yd3 < 1 ? 'bags' : order.yd3 < 5 ? 'compare' : 'ready-mix';
  }
  return out;
}

module.exports = { calcVolume, validate, sectionVolumeFt3, BAG_YIELD_FT3, CONCRETE_DENSITY_LB_FT3 };
```

**H2:** How to Calculate Cubic Yards (Formula and Examples)

> Calculating cubic yards takes three steps: find the volume in cubic feet, divide by 27, then add extra for waste. The same method works for concrete, soil, mulch and gravel.

**H3:** The cubic yard formula

> 1. Convert every measurement to feet (inches ÷ 12).
> 2. **Volume (ft³)** = length × width × depth.
> 3. **Cubic yards** = cubic feet ÷ 27.
> 4. **Order quantity** = cubic yards × (1 + waste %).
>
> Shortcut for a slab: **cubic yards = length (ft) × width (ft) × thickness (in) ÷ 324.**

**H3:** Worked example: 10 × 12 ft patio, 4 in thick (how to estimate concrete yards)

> - Volume: 10 × 12 × (4 ÷ 12) = **40 ft³**
> - Cubic yards: 40 ÷ 27 = **1.48 yd³**
> - Add 10% waste: 1.48 × 1.10 = **1.63 yd³** (44 ft³)
> - To order: about **1.75 yd³** of ready-mix (rounded up to the next quarter yard), or **74 bags** of 80 lb mix (44 ÷ 0.60 = 73.3, rounded up)
> - Weight: about 6,000 lb (3 tons) before waste

**H3:** Figuring yards of concrete by shape

> The formulas below use feet. Convert inches to feet by dividing by 12.

| Shape | Formula | Example |
|---|---|---|
| Slab, patio or driveway | length × width × thickness ÷ 27 | 10 × 12 ft, 4 in thick → 1.48 yd³ |
| Footing or trench | run length × width × depth ÷ 27 | 100 ft, 16 in wide, 8 in deep → 3.29 yd³ |
| Wall | length × height × thickness ÷ 27 | 20 ft long, 4 ft high, 8 in thick → 1.98 yd³ |
| Round column, pier or post hole | π × (diameter ÷ 2)² × height ÷ 27 | 4 columns, 12 in across, 4 ft tall → 0.47 yd³ |

**H2:** How Much Is a Yard of Concrete?

> A yard of concrete means one cubic yard: a volume of 3 ft × 3 ft × 3 ft, or 27 cubic feet. At the usual density of 150 lb/ft³ it weighs about 4,050 lb, a little over 2 tons. You can mix it from roughly 45 bags of 80 lb mix, or order it from a ready-mix plant by the yard.

**H2:** How Many Bags of Concrete Do I Need?

> Bagged mix is sold by weight, but the number of bags you need depends on yield, the amount of mixed concrete one bag makes. Typical yields are below. Always confirm the yield printed on your bag, because it varies by product.

| Bag size | Typical yield per bag | Bags per cubic yard |
|---|---|---|
| 40 lb | 0.30 ft³ (0.011 yd³) | 90 |
| 50 lb | 0.375 ft³ (0.014 yd³) | 72 |
| 60 lb | 0.45 ft³ (0.017 yd³) | 60 |
| 80 lb | 0.60 ft³ (0.022 yd³) | 45 |

> **Bags needed = cubic feet ÷ yield per bag, rounded up.** Add your waste percentage to the volume first. Bags work well for small jobs, but 45 bags per yard is about 3,600 lb of dry mix to lift and mix, which is why larger pours usually go to a ready-mix truck.
