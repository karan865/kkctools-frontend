## B4. Reference calculation engine (tested)

```js
// asphalt-engine.js — framework-independent calculation core (no DOM, no deps)
// Reuse this same pattern for Stone / Sand / Fill Dirt: only the density presets and labels change.

const LB_PER_SHORT_TON = 2000;
const KG_PER_LB = 0.45359237;
const FT_PER_M = 3.280839895013123;

const LENGTH_TO_FT = {
  ft: 1, in: 1 / 12, yd: 3,
  m: FT_PER_M, cm: FT_PER_M / 100, mm: FT_PER_M / 1000,
};
const AREA_TO_SQFT = { sqft: 1, sqyd: 9, sqm: FT_PER_M * FT_PER_M };
const DENSITY_TO_LB_FT3 = {
  'lb/ft3': 1,
  'kg/m3': 1 / 16.01846337396,      // 1 lb/ft3 = 16.01846 kg/m3
  'tons/yd3': LB_PER_SHORT_TON / 27, // short tons per cubic yard
  'lb/yd3': 1 / 27,
};

const toFt = (v, unit) => v * LENGTH_TO_FT[unit];

function sectionAreaSqFt(s) {
  switch (s.shape) {
    case 'rectangle': return toFt(s.length, s.lengthUnit) * toFt(s.width, s.widthUnit);
    case 'circle': { const r = toFt(s.diameter, s.diameterUnit) / 2; return Math.PI * r * r; }
    case 'triangle': return 0.5 * toFt(s.base, s.baseUnit) * toFt(s.height, s.heightUnit);
    case 'area': return s.area * AREA_TO_SQFT[s.areaUnit];
    default: throw new Error('Unknown shape: ' + s.shape);
  }
}

function validate(input) {
  const errors = [];
  const pos = (n) => typeof n === 'number' && isFinite(n) && n > 0;
  if (!input.sections || !input.sections.length) errors.push('Add at least one section.');
  (input.sections || []).forEach((s, i) => {
    const a = (() => { try { return sectionAreaSqFt(s); } catch (e) { return NaN; } })();
    if (!pos(a)) errors.push(`Section ${i + 1}: enter dimensions greater than 0.`);
  });
  const thicknessIn = toFt(input.thickness, input.thicknessUnit) * 12;
  if (!pos(thicknessIn)) errors.push('Enter a thickness greater than 0.');
  else if (thicknessIn > 24) errors.push('Thickness above 24 in (60 cm) is not supported.');
  const dens = input.density * DENSITY_TO_LB_FT3[input.densityUnit];
  if (!pos(dens)) errors.push('Enter a density greater than 0.');
  else if (dens < 60 || dens > 200) errors.push('Density looks out of range (expected roughly 90–160 lb/ft³).');
  if (!(input.wastePct >= 0 && input.wastePct <= 50)) errors.push('Waste must be between 0% and 50%.');
  return errors;
}

function calcAsphalt(input) {
  const errors = validate(input);
  if (errors.length) return { ok: false, errors };

  const areaSqFt = input.sections.reduce((sum, s) => sum + sectionAreaSqFt(s), 0);
  const thicknessFt = toFt(input.thickness, input.thicknessUnit);
  const densityLbFt3 = input.density * DENSITY_TO_LB_FT3[input.densityUnit];
  const wasteFactor = 1 + input.wastePct / 100;

  const volFt3 = areaSqFt * thicknessFt;
  const exactLb = volFt3 * densityLbFt3;
  const exactTons = exactLb / LB_PER_SHORT_TON;           // US short tons
  const orderTons = exactTons * wasteFactor;

  const build = (factor) => ({
    tons: exactTons * factor,
    tonnes: (exactLb * KG_PER_LB / 1000) * factor,         // metric tonnes
    pounds: exactLb * factor,
    cubicFeet: volFt3 * factor,
    cubicYards: (volFt3 / 27) * factor,
    cubicMeters: (volFt3 / (FT_PER_M ** 3)) * factor,
  });

  const out = {
    ok: true,
    areaSqFt, areaSqYd: areaSqFt / 9, areaSqM: areaSqFt / (FT_PER_M ** 2),
    thicknessIn: thicknessFt * 12,
    densityLbFt3,
    densityTonsPerYd3: densityLbFt3 * 27 / LB_PER_SHORT_TON,
    exact: build(1),
    order: build(wasteFactor),
    suggestedOrderTons: Math.ceil(orderTons * 2) / 2,       // round UP to nearest 0.5 ton
    coverageSqFtPerTon: LB_PER_SHORT_TON / (densityLbFt3 * thicknessFt),
  };
  if (input.pricePerTon > 0) out.materialCost = orderTons * input.pricePerTon;
  if (input.truckCapacityTons > 0) out.truckloads = orderTons / input.truckCapacityTons;
  if (input.binderPct > 0) out.binderTons = orderTons * input.binderPct / 100;
  return out;
}

module.exports = { calcAsphalt, validate, sectionAreaSqFt };
```

**H2:** How to Calculate Asphalt Tonnage (Figuring Asphalt Step by Step)

> Figuring asphalt comes down to three numbers: the area you are paving, the compacted thickness, and the density of the mix. Everything else is unit conversion.

**H3:** The asphalt tonnage formula

> 1. **Area (sq ft)** = length × width
> 2. **Volume (ft³)** = area × thickness in inches ÷ 12
> 3. **Weight (lb)** = volume × density (lb/ft³)
> 4. **Tons** = weight ÷ 2,000
>
> Combined: **Tons = area (sq ft) × thickness (in) × density (lb/ft³) ÷ 24,000.** To get cubic yards, divide cubic feet by 27. To get the quantity to order, multiply the tons by 1 plus your waste percentage.

**H3:** Worked example: 20 ft × 40 ft driveway, 2 in thick

> - Area: 20 × 40 = **800 sq ft**
> - Volume: 800 × 2 ÷ 12 = **133.3 ft³** (4.94 yd³)
> - Weight: 133.3 × 145 = **19,333 lb**
> - Tons: 19,333 ÷ 2,000 = **9.67 tons**
> - Add 5% waste: 9.67 × 1.05 = **10.15 tons to order**
>
> Round up to about 10.5 tons and ask your plant about minimum load sizes.

**H2:** Asphalt Density and Weight Reference

> Hot mix asphalt is usually estimated at 145 lb/ft³ once compacted. That is about 3,915 lb, or 1.96 tons, per cubic yard, and roughly 2.32 metric tonnes per cubic meter. Real mixes vary with aggregate, binder content and compaction, so use your supplier's figure when you have one.

| Material | Typical density (lb/ft³) | Tons per cubic yard | kg/m³ |
|---|---:|---:|---:|
| Dense-graded hot mix (planning default) | 145 | 1.96 | 2,323 |
| Stone matrix asphalt (SMA) and warm mix | 140–145 | 1.89–1.96 | 2,243–2,323 |
| Open-graded / porous asphalt | 115–135 | 1.55–1.82 | 1,842–2,162 |
| Asphalt millings (RAP), compacted | about 110 | about 1.49 | about 1,762 |

> A handy shortcut: at 145 lb/ft³, each inch of thickness weighs about 109 lb per square yard, which is the familiar "110 lb per square yard per inch" rule of thumb.

> Thickness moves your tonnage more than any other input, so confirm it before you order. These are common planning ranges for compacted surface course, not design rules. Your contractor, engineer or local agency sets the actual specification.

**H3:** Asphalt for driveways cost: material vs. installed

> Material is only one part of a paved driveway. A contractor's quote for the cost to asphalt a driveway usually also covers excavation, base stone, grading, compaction equipment, trucking, labor and cleanup. That is why an installed price is much higher than tons × price per ton. Get at least three written local quotes and ask what base thickness and asphalt thickness each one includes.

**H3:** Asphalt overlay cost

> An overlay places a new layer, commonly 1.5 to 2 inches, over existing pavement that is still structurally sound. It needs less material than a new driveway and skips the base, so enter the overlay thickness and your area to see the tonnage. Cracks and drainage problems in the old surface usually need repair first, and that work adds to the price.

**H2:** Tips Before You Order

> - **Measure twice.** Add 5–10% for an uneven base, edges and spillage. The calculator starts at 5%.
> - **Ask about minimum loads.** Plants usually sell by the ton and may have minimum load sizes or charge extra for partial loads.
> - **Check the weather window.** Asphalt is commonly laid when the air temperature is above about 50°F (10°C) and rising. Your contractor will advise.
> - **Price the extras separately.** Tack coat, base stone, sealer and labor are not part of this tonnage.

**H2:** Frequently Asked Questions

**H3:** How many tons of asphalt do I need for a driveway?
> It depends on the area and thickness. A 20 × 40 ft driveway (800 sq ft) at 2 inches thick needs about 9.7 tons, or about 10.2 tons with 5% waste. At 3 inches thick, the same driveway needs about 14.5 tons before waste.

**H3:** How much does 1 ton of asphalt cover?
> At 145 lb/ft³, one ton covers about 83 square feet at 2 inches thick, about 55 square feet at 3 inches, and about 41 square feet at 4 inches.

**H3:** How do you figure asphalt tonnage?
> Multiply the area in square feet by the thickness in inches and by the density in lb/ft³, then divide by 24,000. For example, 800 × 2 × 145 ÷ 24,000 = 9.67 tons.

**H3:** How much does a cubic yard of asphalt weigh?
> About 3,915 lb, or 1.96 tons, at 145 lb/ft³. Compacted millings are lighter, around 1.5 tons per cubic yard.

**H3:** Should I add extra for compaction and waste?
> The 145 lb/ft³ figure describes compacted asphalt and thickness should be the compacted thickness, so compaction is already in the math. Add 5–10% for an uneven base, edge spread and spillage. The calculator uses 5% by default.

**H3:** How thick should an asphalt driveway be?
> Most new residential driveways use about 2–3 inches of compacted asphalt over a prepared stone base. Heavier vehicles may need more. Confirm with your contractor or local requirements.

**H3:** How much does it cost to asphalt a driveway?
> Material cost is the tons you order times your supplier's price per ton. A contractor's installed price is higher because it also includes excavation, base, grading, equipment, trucking and labor. Compare at least three written quotes.

**H3:** What is the cost of an asphalt overlay?
> It depends on the area, the overlay thickness (commonly 1.5–2 inches), your local price per ton, and any crack or drainage repairs needed first. Enter your area and thickness to get the tonnage, then add your price per ton.

**H3:** Can I use this as an asphalt millings calculator?
> Yes. Select Millings to use a default density of 110 lb/ft³. Because millings weight varies with moisture and gradation, replace it with your supplier's figure when you have one.

**H3:** Is bitumen the same as asphalt?
> Bitumen is the sticky binder. Asphalt is the finished mix of bitumen and aggregate such as stone and sand. In everyday use the two words overlap, which is why this calculator can estimate both the mix and the binder in it.
